\section*{Sintesi}

\begin{itemize}
  \item \textbf{Il guadagno è reale e non è spiegato dal numero di parametri.} Un singolo blocco V-Mamba dopo \cfg{out1} ($64\times64$) aggiunge solo \textbf{+1,2\% di parametri} e riduce la validation loss del \textbf{$\sim$9\%} rispetto alla baseline. Su tutti i modelli la correlazione tra parametri e validation loss è \textbf{nulla} (Pearson $r \approx 0{,}00$). Nello sweep di posizione, anzi, \textbf{più parametri $\rightarrow$ loss peggiore}.
  \item \textbf{I migliori:} \cfg{MMM\_AA} e \cfg{halfMamba} ($-11\%$ di val loss) e \cfg{M\_out1} ($-9\%$ con quasi zero parametri in più). Tutte le migliori configurazioni hanno \textbf{Mamba ad alta risoluzione (\cfg{out1})}.
  \item \textbf{I peggiori:} Mamba o attention nel \textbf{bottleneck} (\cfg{M\_out4}, \cfg{mambaSup}, attention bottleneck originale) danno un miglioramento \textbf{indistinguibile dal rumore}. Un blocco di attention a \cfg{out3} \textbf{peggiora} le architetture ibride.
  \item \textbf{Limite principale:} quasi tutti i risultati sono a \textbf{1 seed} e solo sulla \textbf{validazione}. Il test set è stato valutato solo per la baseline. Le conclusioni vanno confermate con 3 seed e con il test (Sez.~\ref{sec:futuri}).
  \item L'errore sulla \textbf{pressione} ($\sim$9,5\% val, $\sim$14,5\% test) è la componente dominante e \textbf{nessuna architettura la migliora in modo significativo}: i guadagni sono su $v_x$ e soprattutto $v_y$.
\end{itemize}

\subsection*{Modifiche apportate al codice (storia git)}

\begin{enumerate}
  \item \textbf{\cfg{TurbNet} configurabile} (\cfg{DfpNet.py}): la U-Net originale con ``mixer'' opzionali dopo ogni stage dell'encoder. Il mixer sostituisce l'output dello stage sia verso il layer successivo sia nella skip connection. I tipi di mixer sono:
  \begin{itemize}
    \item \cfg{vmamba\_big}: VSSBlock con d\_state 16, ssm\_ratio 2, mlp\_ratio 4;
    \item \cfg{vmamba\_small}: d\_state 8, ssm\_ratio 1, mlp\_ratio 2;
    \item \cfg{attention}: MHSA con 8 teste, LayerNorm e residual.
  \end{itemize}
  \item \textbf{Kernel della conv locale di Mamba (\cfg{ssm\_conv})}: $3 \rightarrow 2 \rightarrow 1$ (commit \cfg{de207ab}, \cfg{a128172} del 14/10). \textbf{Attenzione:} \cfg{mambaSup\_exp1} è stato allenato \emph{prima} di questa modifica (presumibilmente \cfg{ssm\_conv=3}, 2.421.187 parametri contro 2.419.907 di oggi). Non è quindi perfettamente confrontabile con \cfg{M\_out2b}, che oggi ha la stessa posizione.
  \item Architetture aggiunte nel tempo: \cfg{mambaSup}, \cfg{fullMamba}, \cfg{halfMamba}, \cfg{halfMamba\_lightconv1}, \cfg{halfMambaPush}, \cfg{MMM\_AA}, \cfg{MMAA\_\_}, \cfg{MMMA\_\_}.
  \item \textbf{Sweep di posizione} (nuovo): \cfg{M\_out1}\dots\cfg{M\_out5}, un solo blocco \cfg{vmamba\_big} per run (\cfg{run\_mamba\_position.sh}). Lo stesso sweep con attention, \cfg{A\_out1}\dots\cfg{A\_out5} (\cfg{run\_attention\_position.sh}), \textbf{è in corso}.
\end{enumerate}

\section{Migliori e peggiori}

\subsection*{Migliori}
\begin{enumerate}
  \item \textbf{\cfg{MMM\_AA} / \cfg{halfMamba\_lightconv1}}: val loss 0,00324 ($-11{,}4\%$), errori più bassi su $v_x$ ($-10\%$) e $v_y$ ($-12\%$).
  \item \textbf{\cfg{halfMamba}, \cfg{halfMambaPush}, \cfg{M\_out1}}: tutti tra $-9\%$ e $-10\%$.
  \item \textbf{Miglior rapporto prestazioni/costo: \cfg{M\_out1}.} Ottiene circa l'80\% del guadagno di \cfg{MMM\_AA} con +28k parametri invece di +732k. È anche la più veloce tra le top-5 (15\,s/epoca contro 21--25).
\end{enumerate}

\cfg{halfMamba} e \cfg{halfMamba\_lightconv1}, con la stessa configurazione nel codice attuale, differiscono di 0,00005. È una stima diretta del rumore single-seed, e i primi 5 della classifica sono \textbf{statisticamente indistinguibili tra loro}.

\subsection*{Peggiori}
\begin{enumerate}
  \item \textbf{Blocchi nel bottleneck:} \cfg{M\_out4} ($-1{,}6\%$ con \textbf{+47\%} di parametri), \cfg{mambaSup} ($-1{,}7\%$) e l'attention bottleneck originale ($\sim$0\%). Sono tutti nel range del rumore.
  \item \textbf{Attention a \cfg{out3} nelle ibride:}
  \begin{itemize}
    \item \cfg{MMMA\_\_} = \cfg{halfMamba} + Attn@out3 $\rightarrow$ da $-10{,}0\%$ a $\mathbf{-2{,}4\%}$;
    \item \cfg{MMAA\_\_} = \cfg{halfMambaPush} + Attn@out3 $\rightarrow$ da $-9{,}5\%$ a $-4{,}6\%$.
  \end{itemize}
  Aggiungere attention a \cfg{out3} \textbf{distrugge} gran parte del guadagno.
  \item \textbf{\cfg{fullMamba}:} +57\% di parametri, la più lenta (35\,s/epoca), e peggiore di \cfg{halfMamba}. I blocchi Mamba profondi (\cfg{vmamba\_small} da out2b a out5) non aiutano, anzi peggiorano.
\end{enumerate}

\begin{itemize}
  \item \textbf{Lo stage conta molto più dei parametri.} Solo a \cfg{out1} il guadagno è chiaramente sopra il rumore. Da \cfg{out2} in giù tutte le posizioni stanno in una fascia di $-2/-5\%$, compatibile con la variabilità tra seed.

\section{Miglioramento reale o solo più parametri?}

\textbf{Conclusione: il miglioramento è architetturale, non dovuto alla capacità.} Le evidenze:

\begin{enumerate}
  \item \textbf{Nessuna correlazione tra parametri e loss su tutti i 14 modelli.} Pearson $r = -0{,}004$ ($p = 0{,}99$), Spearman $\rho = -0{,}15$ ($p = 0{,}62$).
  \item \textbf{Nello sweep di posizione la correlazione è al contrario} (Spearman $\rho = +0{,}72$, cioè più parametri $\rightarrow$ loss più alta). Le posizioni più costose (\cfg{out4}/\cfg{out5}, +1,09\,M parametri, +47\%) sono tra le peggiori, mentre la più economica (\cfg{out1}, +28k) è la migliore.
  \item \textbf{Confronti a parità (quasi) di parametri} (Tabella~\ref{tab:parita}).
\end{enumerate}

\begin{enumerate}[resume]
  \item \textbf{Vincolo di iso-capacità (README):} \cfg{M\_out1} (+1,2\%), \cfg{halfMambaPush} (+5,7\%) e \cfg{halfMamba} (+8,8\%) lo rispettano e restano tra i migliori. \cfg{MMM\_AA} (+31\%) e \cfg{fullMamba} (+57\%) \textbf{no}. Quindi il ``primo posto'' di \cfg{MMM\_AA} non va considerato una vittoria pulita.
\end{enumerate}

\paragraph{Cosa manca per una prova definitiva.} Un \textbf{controllo a parità di parametri senza Mamba}. Esempi: baseline leggermente più larga, oppure baseline con un blocco convoluzionale residuo a \cfg{out1} da $\sim$28k parametri (Sez.~\ref{sec:futuri}, punto 2). Se questo controllo non raggiunge il $-9\%$, il beneficio è attribuibile alla scansione selettiva e non alla capacità né alla profondità extra.

\subsection*{Rumore e significatività}
\begin{itemize}
  \item Deviazione standard tra seed: baseline 0,000023 (0,6\%), \cfg{mambaSup} 0,00014 (3,9\%). La stima della baseline è probabilmente ottimistica, perché 3 seed sono pochi.
  \item Adottando un margine prudente di $\pm4\%$:
  \begin{itemize}
    \item guadagni $\geq 9\%$ (\cfg{out1}, \cfg{halfMamba*}, \cfg{MMM\_AA}): \textbf{probabilmente reali};
    \item guadagni $\leq 5\%$ (sweep da out2 a out5, \cfg{MMAA\_\_}, \cfg{MMMA\_\_}, \cfg{mambaSup}): \textbf{non distinguibili dal rumore}.
  \end{itemize}
  \item Il guadagno è concentrato sulla \textbf{velocità} ($v_y$ da 4,31 a 3,77--3,98\%, cioè $\sim3\sigma$; $v_x$ da 1,10 a 0,99\%). Sulla \textbf{pressione} il migliore (9,37\%) è entro una deviazione standard dalla baseline ($9{,}60 \pm 0{,}41\%$).
\end{itemize}

\section{Costo computazionale}
\label{sec:costo}

\begin{itemize}
  \item \textbf{Mamba scala bene in alta risoluzione:} \cfg{M\_out1} costa 15\,s/epoca e 386\,MB.
  \item \textbf{L'attention in alta risoluzione no:} \cfg{A\_out1} costa 105\,s/epoca ($\times7$) e 10,3\,GB ($\times27$).
  \item \textbf{Causa verificata:} in AMP (fp16), con 32 canali / 8 teste, \cfg{head\_dim = 4}. Le kernel efficienti di \cfg{scaled\_dot\_product\_attention} non vengono usate e PyTorch materializza l'intera matrice di attention $4096\times4096$. Un test isolato in fp32 usava solo $\sim$250\,MB.
\end{itemize}

Per i confronti futuri con l'attention ad alta risoluzione va sistemato questo punto (Sez.~\ref{sec:futuri}, punto 6), altrimenti il confronto di efficienza Mamba vs attention è distorto.

\section{Sviluppi futuri}
\label{sec:futuri}

Ordinati per priorità:

\begin{enumerate}
  \item \textbf{Confermare i risultati.}
  \begin{itemize}
    \item Rilanciare con 3 seed (42/1955/2003) almeno \cfg{M\_out1}, \cfg{halfMamba}, \cfg{halfMambaPush} e \cfg{MMM\_AA} (con \cfg{run\_all\_seeds.py}).
    \item Valutarli sul \textbf{test set} con \cfg{runTest.py}, dato che oggi è stato testato solo la baseline.
    \item Senza questo passo le differenze tra i top-5 non sono interpretabili.
  \end{itemize}
  \item \textbf{Controllo a parità di parametri} (chiude la domanda della sezione precedente):
  \begin{itemize}
    \item \cfg{baseline} con \cfg{expo} non intero, dato che \cfg{channels = int(2**expo + 0.5)} lo permette (es. 33--34 canali);
    \item oppure una baseline con un blocco conv $3\times3$ residuo a \cfg{out1} con $\sim$28k parametri, come \cfg{M\_out1}.
  \end{itemize}
  \item \textbf{Spingere ancora più in alta risoluzione}, visto che la risoluzione è il fattore chiave.
  \begin{itemize}
    \item Il codice supporta già lo stage \cfg{in} ($128\times128$) ma non è mai stato testato. A 3 canali il blocco è minuscolo, quindi probabilmente conviene una piccola stem conv prima.
    \item Valutare Mamba anche nel \textbf{decoder} (es.\ su \cfg{dout2} o prima di \cfg{dlayer1}), dove si ricostruisce la scia.
  \end{itemize}
  \item \textbf{Scansioni non raster a \cfg{out1} (RQ6):} curva di Hilbert o scansione streamwise. Ha senso farlo proprio a \cfg{out1}, dove Mamba ha effetto. Lì le sequenze sono lunghe e la discontinuità della scansione raster pesa di più.
  \item \textbf{Alleggerire i blocchi profondi.}
  \begin{itemize}
    \item \cfg{vmamba\_big} a 256 canali (\cfg{out4}/\cfg{out5}) costa 0,54\,M parametri, soprattutto per l'MLP con ratio 4, senza benefici.
    \item Se servono mixer profondi, usare \cfg{vmamba\_small} o \cfg{mlp\_ratio} ridotto. In generale conviene evitare attention a \cfg{out3}.
  \end{itemize}
  \item \textbf{Sistemare l'attention per i confronti equi.} Usare \cfg{num\_heads = 4} a 32 canali (\cfg{head\_dim = 8}, che abilita le kernel efficienti in fp16), oppure tenere l'attention in fp32 o bf16. Va fatto prima di trarre conclusioni dallo sweep \cfg{A\_out*}.
  \item \textbf{Attaccare l'errore sulla pressione}, oggi il collo di bottiglia ($\sim$9,5\% val, $\sim$14,5\% test):
  \begin{itemize}
    \item pesare di più il canale $p$ nella loss;
    \item normalizzazione adimensionale (opzione già presente in \cfg{dataset.py});
    \item loss con gradienti o termini fisici (es.\ divergenza della velocità).
  \end{itemize}
  \item \textbf{Training e protocollo.}
  \begin{itemize}
    \item Salvare il \textbf{checkpoint migliore} in validazione: oggi si salva l'ultimo, e diverse run hanno la best loss qualche epoca prima della fine.
    \item Registrare tempo/epoca e VRAM anche per la baseline, che manca.
    \item Allenare il modello finale sul \textbf{dataset completo} (README, Sez.~4).
  \end{itemize}
  \item \textbf{Pulizia e riproducibilità.}
  \begin{itemize}
    \item I risultati di \cfg{fullMamba} sono in \cfg{metrics/metrics\_train\_fullMamba.csv} invece che nella sua cartella, dove c'è un CSV vuoto. Lo stesso vale per \cfg{mambaSup\_debug}.
    \item Riallenare \cfg{mambaSup} con \cfg{ssm\_conv=1} per allinearlo agli altri modelli.
    \item Annotare in \cfg{MODEL\_CONFIGS} gli iperparametri di ogni run storica (es.\ cosa distingue \cfg{halfMamba\_lightconv1} da \cfg{halfMamba\_exp1}: dal codice attuale non si ricostruisce).
  \end{itemize}
\end{enumerate}
